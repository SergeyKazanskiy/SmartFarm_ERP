1. Провести приход с пересчётом скользящего среднего.

python
from decimal import Decimal
from apps.companies.models import Company
from apps.wms.models import Inventory
from apps.wms.services import (
    MaterialService, StorageLocationService, WarehouseService,
    GoodsReceiptService, GoodsReceiptLineService, WarehouseTypeService,
)

company = Company.objects.first()

wt = WarehouseTypeService.create(company_id=company.id, data={"name": "Склад"})
wh = WarehouseService.save(company_id=company.id, data={
    "name": "Центральный", "warehouse_type_id": wt.id,
})
loc = StorageLocationService.save(company_id=company.id, data={
    "warehouse_id": wh.id, "location_code": "A-01-01",
})
mat = MaterialService.save(company_id=company.id, data={
    "material_code": "PCB-001", "name": "Печатная плата",
})

# Первый приход: 1000 шт × 145
gr1 = GoodsReceiptService.create(company_id=company.id, data={
    "warehouse_id": wh.id, "receipt_date": "2026-06-01T10:00:00Z",
})
GoodsReceiptLineService.create(company_id=company.id, data={
    "receipt_id": gr1.id, "material_id": mat.id,
    "location_id": loc.id, "quantity": 1000, "unit_cost": 145,
})
GoodsReceiptService.post(company_id=company.id, receipt_id=gr1.id)

inv = Inventory.objects.get(company_id=company.id, material_id=mat.id)
print("After 1st receipt: qty=", inv.quantity, "cost=", inv.unit_cost)
# qty=1000, cost=145.0000

# Второй приход: 500 шт × 160
gr2 = GoodsReceiptService.create(company_id=company.id, data={
    "warehouse_id": wh.id, "receipt_date": "2026-06-10T10:00:00Z",
})
GoodsReceiptLineService.create(company_id=company.id, data={
    "receipt_id": gr2.id, "material_id": mat.id,
    "location_id": loc.id, "quantity": 500, "unit_cost": 160,
})
GoodsReceiptService.post(company_id=company.id, receipt_id=gr2.id)

inv.refresh_from_db()
print("After 2nd receipt: qty=", inv.quantity, "cost=", inv.unit_cost)
# qty=1500, cost = (1000×145 + 500×160) / 1500 = 150.0000
